cmake_minimum_required(VERSION 3.20)
project(NRColorTests LANGUAGES CXX)
enable_testing()
add_executable(nr_color_policy_test policy_test.cpp)
target_compile_features(nr_color_policy_test PRIVATE cxx_std_20)
target_include_directories(nr_color_policy_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
add_test(NAME NRColorPolicy COMMAND nr_color_policy_test)
add_executable(nr_exposure_policy_test exposure_policy_test.cpp)
target_compile_features(nr_exposure_policy_test PRIVATE cxx_std_20)
target_include_directories(nr_exposure_policy_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
add_test(NAME NRExposurePolicy COMMAND nr_exposure_policy_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME NRColorSourceContracts
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/source_contract_test.py")
    add_test(NAME NRColorAssessmentContracts
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/assessment_test.py")
    add_test(NAME NRColorAssetInventory
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../tools/nr-color/verify_assets.py")
endif()
if(WIN32)
    foreach(_test IN ITEMS shader_gpu exposure_shader_gpu)
        add_executable(nr_color_${_test}_test ${_test}_test.cpp)
        target_compile_features(nr_color_${_test}_test PRIVATE cxx_std_20)
        target_compile_definitions(nr_color_${_test}_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_include_directories(nr_color_${_test}_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
        target_link_libraries(nr_color_${_test}_test PRIVATE d3d11 d3dcompiler)
        if(_test STREQUAL "shader_gpu")
            set(_name NRColorShadersWARP)
        else()
            set(_name NRColorExposureWARP)
        endif()
        add_test(NAME ${_name} COMMAND nr_color_${_test}_test
            "${CMAKE_CURRENT_LIST_DIR}/../../features/Neural Rendering Colour/Shaders/Upscaling/NeuralRendering")
    endforeach()
endif()
