cmake_minimum_required(VERSION 3.20)
project(NRColorTests LANGUAGES CXX)
enable_testing()
add_executable(nr_color_policy_test policy_test.cpp)
target_compile_features(nr_color_policy_test PRIVATE cxx_std_20)
target_include_directories(nr_color_policy_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
add_test(NAME NRColorPolicy COMMAND nr_color_policy_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME NRColorSourceContracts
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/source_contract_test.py")
endif()
if(WIN32)
    add_executable(nr_color_shader_gpu_test shader_gpu_test.cpp)
    target_compile_features(nr_color_shader_gpu_test PRIVATE cxx_std_20)
    target_compile_definitions(nr_color_shader_gpu_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(nr_color_shader_gpu_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
    target_link_libraries(nr_color_shader_gpu_test PRIVATE d3d11 d3dcompiler)
    add_test(NAME NRColorShadersWARP COMMAND nr_color_shader_gpu_test
        "${CMAKE_CURRENT_LIST_DIR}/../../features/Neural Rendering Colour/Shaders/Upscaling/NeuralRendering")
endif()
