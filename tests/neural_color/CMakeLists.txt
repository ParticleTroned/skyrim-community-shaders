cmake_minimum_required(VERSION 3.21)
project(NRColorTests LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(nr_color_policy policy_test.cpp)
target_include_directories(nr_color_policy PRIVATE "${REPO_ROOT}/src/Features/Upscaling/NeuralRendering")
if(MSVC)
    target_compile_options(nr_color_policy PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(nr_color_policy PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME NRColorPolicy COMMAND nr_color_policy)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME NRColorIntegrationContract
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/integration_test.py" "${REPO_ROOT}")
endif()
if(WIN32)
    add_executable(nr_color_gpu gpu_test.cpp)
    target_link_libraries(nr_color_gpu PRIVATE d3d11 d3dcompiler)
    add_test(NAME NRColorGPU COMMAND nr_color_gpu
        "${REPO_ROOT}/features/Upscaling/Shaders/Upscaling/NeuralRendering/ColorPipelineCS.hlsl")
endif()
